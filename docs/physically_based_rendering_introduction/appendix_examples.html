149def dot(a: FloatArray, b: FloatArray) -> FloatArray:
150    """行ごとの内積を返す。"""
151    result: FloatArray = np.einsum("ij,ij->i", a, b)
152    return result

155def normalize(v: FloatArray) -> FloatArray:
156    """行ごとに長さ 1 に正規化する。"""
157    result: FloatArray = v / np.linalg.norm(v, axis=-1, keepdims=True)
158    return result

161def orthonormal_basis(normals: FloatArray) -> tuple[FloatArray, FloatArray]:
162    """法線から、法線と直交する 2 本の単位ベクトル（接線と従接線）を作る。
163
164    Duff らの方法（"Building an Orthonormal Basis, Revisited", 2017）による。
165    """
166    nx, ny, nz = normals[:, 0], normals[:, 1], normals[:, 2]
167    sign = np.where(nz >= 0.0, 1.0, -1.0)
168    a = -1.0 / (sign + nz)
169    b = nx * ny * a
170    tangent = np.stack([1.0 + sign * nx * nx * a, sign * b, -sign * nx], axis=1)
171    bitangent = np.stack([b, sign + ny * ny * a, -ny], axis=1)
172    return tangent, bitangent
173
174
175def to_world(local: FloatArray, normals: FloatArray) -> FloatArray:
176    """法線を z 軸とする局所座標系のベクトルをワールド座標系に変換する。"""
177    tangent, bitangent = orthonormal_basis(normals)
178    result: FloatArray = local[:, 0:1] * tangent + local[:, 1:2] * bitangent + local[:, 2:3] * normals
179    return result

182def rectangle_normal(rect: Rectangle) -> FloatArray:
183    """長方形の表側の単位法線を返す。"""
184    normal = np.cross(np.array(rect.edge_u), np.array(rect.edge_v))
185    result: FloatArray = normal / np.linalg.norm(normal)
186    return result

193def intersect_sphere(sphere: Sphere, origins: FloatArray, directions: FloatArray) -> FloatArray:
194    """各レイが球と最初に交わる距離 t を返す。交わらない場合は inf とする。"""
195    oc = origins - np.array(sphere.center)
196    b = dot(oc, directions)
197    c = dot(oc, oc) - sphere.radius ** 2
198    discriminant = b * b - c
199    sqrt_d = np.sqrt(np.maximum(discriminant, 0.0))
200    t_near = -b - sqrt_d
201    t_far = -b + sqrt_d
202    t = np.where(t_near > EPSILON, t_near, t_far)
203    hit = (discriminant > 0.0) & (t > EPSILON)
204    result: FloatArray = np.where(hit, t, np.inf)
205    return result
206
207
208def intersect_rectangle(rect: Rectangle, origins: FloatArray, directions: FloatArray) -> FloatArray:
209    """各レイが長方形と交わる距離 t を返す。交わらない場合は inf とする。"""
210    corner = np.array(rect.corner)
211    edge_u = np.array(rect.edge_u)
212    edge_v = np.array(rect.edge_v)
213    normal = rectangle_normal(rect)
214    with np.errstate(divide="ignore", invalid="ignore"):
215        denom = directions @ normal
216        t = ((corner - origins) @ normal) / denom
217        rel = origins + t[:, None] * directions - corner
218        u = rel @ edge_u / (edge_u @ edge_u)
219        v = rel @ edge_v / (edge_v @ edge_v)
220        hit = (np.abs(denom) > 1e-12) & (t > EPSILON) & (u >= 0.0) & (u <= 1.0) & (v >= 0.0) & (v <= 1.0)
221    result: FloatArray = np.where(hit, t, np.inf)
222    return result

225def intersect_scene(scene: Scene, origins: FloatArray, directions: FloatArray) -> tuple[FloatArray, IntArray]:
226    """最も近い交差点までの距離と、交差した物体の番号（交差なしは -1）を返す。"""
227    nearest_t = np.full(len(origins), np.inf)
228    shape_ids = np.full(len(origins), -1, dtype=np.intp)
229    for index, shape in enumerate(scene.shapes):
230        if isinstance(shape, Sphere):
231            t = intersect_sphere(shape, origins, directions)
232        else:
233            t = intersect_rectangle(shape, origins, directions)
234        closer = t < nearest_t
235        nearest_t[closer] = t[closer]
236        shape_ids[closer] = index
237    return nearest_t, shape_ids

240def normals_at(scene: Scene, shape_ids: IntArray, points: FloatArray) -> FloatArray:
241    """交差点における物体の幾何法線（表側）を返す。"""
242    normals = np.zeros_like(points)
243    for index, shape in enumerate(scene.shapes):
244        mask = shape_ids == index
245        if not np.any(mask):
246            continue
247        if isinstance(shape, Sphere):
248            normals[mask] = (points[mask] - np.array(shape.center)) / shape.radius
249        else:
250            normals[mask] = rectangle_normal(shape)
251    return normals

258def ggx_distribution(cos_h: FloatArray, alpha: FloatArray) -> FloatArray:
259    """GGX（Trowbridge-Reitz）法線分布関数 D。"""
260    a2 = alpha * alpha
261    denom = cos_h * cos_h * (a2 - 1.0) + 1.0
262    result: FloatArray = a2 / (math.pi * denom * denom)
263    return result
264
265
266def smith_g1(cos_v: FloatArray, alpha: FloatArray) -> FloatArray:
267    """GGX に対する Smith のマスキング関数 G1。"""
268    a2 = alpha * alpha
269    result: FloatArray = 2.0 * cos_v / (cos_v + np.sqrt(a2 + (1.0 - a2) * cos_v * cos_v))
270    return result
271
272
273def fresnel_schlick(cos_theta: FloatArray, f0: FloatArray) -> FloatArray:
274    """フレネル反射率の Schlick 近似。f0 は (N, 3) の RGB 値。"""
275    result: FloatArray = f0 + (1.0 - f0) * ((1.0 - cos_theta) ** 5)[:, None]
276    return result

279def evaluate_brdf(table: MaterialTable, ids: IntArray, normals: FloatArray,
280                  wo: FloatArray, wi: FloatArray) -> FloatArray:
281    """BRDF の値 f_r(wi, wo) を RGB で返す。wi は表側を向いていることを前提とする。"""
282    albedo = table.albedo[ids]
283    result: FloatArray = albedo / math.pi  # ランバート BRDF
284    metal = table.metal[ids]
285    if np.any(metal):
286        n, o, i = normals[metal], wo[metal], wi[metal]
287        alpha = np.maximum(table.roughness[ids][metal] ** 2, 1e-3)
288        h = normalize(o + i)
289        cos_o = np.maximum(dot(n, o), 1e-6)
290        cos_i = np.maximum(dot(n, i), 1e-6)
291        cos_h = np.clip(dot(n, h), 0.0, 1.0)
292        cos_oh = np.clip(dot(o, h), 0.0, 1.0)
293        d = ggx_distribution(cos_h, alpha)
294        g = smith_g1(cos_o, alpha) * smith_g1(cos_i, alpha)
295        f = fresnel_schlick(cos_oh, albedo[metal])
296        result[metal] = f * (d * g / (4.0 * cos_o * cos_i))[:, None]
297    return result

300def brdf_pdf(table: MaterialTable, ids: IntArray, normals: FloatArray,
301             wo: FloatArray, wi: FloatArray) -> FloatArray:
302    """sample_brdf が方向 wi を選ぶ確率密度（立体角に関する密度）を返す。"""
303    cos_i = dot(normals, wi)
304    result: FloatArray = np.maximum(cos_i, 0.0) / math.pi  # コサイン重み付きサンプリング
305    metal = table.metal[ids]
306    if np.any(metal):
307        n, o, i = normals[metal], wo[metal], wi[metal]
308        alpha = np.maximum(table.roughness[ids][metal] ** 2, 1e-3)
309        h = normalize(o + i)
310        cos_h = np.clip(dot(n, h), 0.0, 1.0)
311        cos_oh = np.maximum(dot(o, h), 1e-6)
312        # ハーフベクトルの確率密度 D(h) cosθ_h を、反射方向の確率密度に変換する
313        pdf = ggx_distribution(cos_h, alpha) * cos_h / (4.0 * cos_oh)
314        result[metal] = np.where(cos_i[metal] > 0.0, pdf, 0.0)
315    return result

318def sample_cosine_hemisphere(normals: FloatArray, rng: np.random.Generator) -> FloatArray:
319    """法線まわりの半球から、確率密度 cosθ / π で方向をサンプリングする。"""
320    xi = rng.random((len(normals), 2))
321    r = np.sqrt(xi[:, 0])
322    phi = 2.0 * math.pi * xi[:, 1]
323    local = np.stack([r * np.cos(phi), r * np.sin(phi), np.sqrt(1.0 - xi[:, 0])], axis=1)
324    return to_world(local, normals)
325
326
327def sample_ggx_half_vector(normals: FloatArray, alpha: FloatArray, rng: np.random.Generator) -> FloatArray:
328    """確率密度 D(h) cosθ_h でハーフベクトル h をサンプリングする。"""
329    xi = rng.random((len(normals), 2))
330    cos_theta = np.sqrt((1.0 - xi[:, 0]) / (1.0 + (alpha * alpha - 1.0) * xi[:, 0]))
331    sin_theta = np.sqrt(np.maximum(0.0, 1.0 - cos_theta * cos_theta))
332    phi = 2.0 * math.pi * xi[:, 1]
333    local = np.stack([sin_theta * np.cos(phi), sin_theta * np.sin(phi), cos_theta], axis=1)
334    return to_world(local, normals)

337def sample_brdf(table: MaterialTable, ids: IntArray, normals: FloatArray, wo: FloatArray,
338                rng: np.random.Generator) -> tuple[FloatArray, FloatArray, FloatArray]:
339    """次の方向 wi、重み f_r cosθ_i / pdf、確率密度 pdf を返す。無効なサンプルの重みは 0 とする。"""
340    albedo = table.albedo[ids]
341    metal = table.metal[ids]
342    diffuse = ~metal
343    wi = np.empty_like(wo)
344    weight = np.zeros_like(wo)
345
346    if np.any(diffuse):
347        # ランバート BRDF では f_r cosθ / pdf = (ρ / π) cosθ / (cosθ / π) = ρ となる
348        wi[diffuse] = sample_cosine_hemisphere(normals[diffuse], rng)
349        weight[diffuse] = albedo[diffuse]
350
351    if np.any(metal):
352        n, o = normals[metal], wo[metal]
353        alpha = np.maximum(table.roughness[ids][metal] ** 2, 1e-3)
354        h = sample_ggx_half_vector(n, alpha, rng)
355        cos_oh = dot(o, h)
356        i = 2.0 * cos_oh[:, None] * h - o  # wo を h に関して反射させる
357        cos_o = dot(n, o)
358        cos_i = dot(n, i)
359        cos_h = dot(n, h)
360        valid = (cos_oh > 0.0) & (cos_i > 0.0) & (cos_o > 0.0)
361        # f_r cosθ_i / pdf = F G (wo・h) / ((n・wo)(n・h))
362        g = smith_g1(np.maximum(cos_o, 1e-6), alpha) * smith_g1(np.maximum(cos_i, 1e-6), alpha)
363        f = fresnel_schlick(np.clip(cos_oh, 0.0, 1.0), albedo[metal])
364        w = f * (g * cos_oh / (np.maximum(cos_o, 1e-6) * np.maximum(cos_h, 1e-6)))[:, None]
365        wi[metal] = i
366        weight[metal] = np.where(valid[:, None], w, 0.0)
367
368    return wi, weight, brdf_pdf(table, ids, normals, wo, wi)

375def light_pdf(scene: Scene, dist2: FloatArray, cos_light: FloatArray) -> FloatArray:
376    """光源上の点を一様に選んだときの、立体角に関する確率密度 r^2 / (cosθ_l A) を返す。"""
377    light = scene.shapes[scene.light_index]
378    assert isinstance(light, Rectangle)
379    area = float(np.linalg.norm(np.cross(light.edge_u, light.edge_v)))
380    result: FloatArray = dist2 / (np.maximum(cos_light, 1e-6) * area)
381    return result

384def mis_weight(pdf_a: FloatArray, pdf_b: FloatArray) -> FloatArray:
385    """多重重点サンプリングのバランスヒューリスティック p_a / (p_a + p_b)。"""
386    total = pdf_a + pdf_b
387    result: FloatArray = np.where(total > 0.0, pdf_a / np.maximum(total, 1e-300), 0.0)
388    return result

391def direct_lighting(scene: Scene, table: MaterialTable, ids: IntArray, points: FloatArray,
392                    normals: FloatArray, wo: FloatArray, rng: np.random.Generator) -> FloatArray:
393    """次イベント推定：光源上の点を 1 つ選び、そこから直接届く光の寄与を求める。
394
395    BRDF のサンプリングで光源に当たる場合の寄与と、多重重点サンプリングで組み合わせる。
396    """
397    light = scene.shapes[scene.light_index]
398    assert isinstance(light, Rectangle)
399    corner = np.array(light.corner)
400    edge_u = np.array(light.edge_u)
401    edge_v = np.array(light.edge_v)
402    light_normal = rectangle_normal(light)
403    area = float(np.linalg.norm(np.cross(edge_u, edge_v)))
404
405    # 光源上の点を一様に選ぶ（面積に関する確率密度は 1 / area）
406    xi = rng.random((len(points), 2))
407    light_points = corner + xi[:, 0:1] * edge_u + xi[:, 1:2] * edge_v
408    to_light = light_points - points
409    dist2 = dot(to_light, to_light)
410    wi = to_light / np.sqrt(dist2)[:, None]
411    cos_surface = dot(normals, wi)
412    cos_light = -(wi @ light_normal)
413
414    # 遮蔽判定（シャドウレイ）：最初に当たる物体が光源なら見えている
415    _, hit_ids = intersect_scene(scene, points + normals * EPSILON, wi)
416    visible = (cos_surface > 0.0) & (cos_light > 0.0) & (hit_ids == scene.light_index)
417
418    brdf = evaluate_brdf(table, ids, normals, wo, wi)
419    emission = table.emission[scene.light_index]
420    geometry = np.where(visible, cos_surface * cos_light * area / dist2, 0.0)
421    weight = mis_weight(light_pdf(scene, dist2, cos_light), brdf_pdf(table, ids, normals, wo, wi))
422    result: FloatArray = brdf * emission * (geometry * weight)[:, None]
423    return result

426def trace_paths(scene: Scene, table: MaterialTable, origins: FloatArray, directions: FloatArray,
427                max_depth: int, use_nee: bool, rng: np.random.Generator) -> FloatArray:
428    """各カメラレイについてパスを 1 本ずつ追跡し、運ばれてきた放射輝度を返す。"""
429    radiance = np.zeros_like(origins)
430    throughput = np.ones_like(origins)       # ここまでのパスの重み（BRDF × cos / pdf の積）
431    pixel = np.arange(len(origins))           # 各パスがどの画素に属するか
432    previous_pdf = np.zeros(len(origins))     # 直前の反射で方向を選んだときの確率密度
433
434    for depth in range(max_depth):
435        t, ids = intersect_scene(scene, origins, directions)
436        hit = ids >= 0                        # 何にも当たらなかったパスは終了
437        pixel, throughput, origins, directions = pixel[hit], throughput[hit], origins[hit], directions[hit]
438        t, ids, previous_pdf = t[hit], ids[hit], previous_pdf[hit]
439        if len(pixel) == 0:
440            break
441
442        points = origins + t[:, None] * directions
443        geometric_normals = normals_at(scene, ids, points)
444        front = dot(geometric_normals, directions) < 0.0
445        normals = np.where(front[:, None], geometric_normals, -geometric_normals)
446        wo = -directions
447
448        # 光源に当たった場合の寄与
449        emitted = throughput * table.emission[ids] * front[:, None]
450        if depth == 0 or not use_nee:
451            radiance[pixel] += emitted
452        else:
453            # 次イベント推定でも同じ経路を数えているので、多重重点サンプリングの重みを掛けて加える
454            pdf_light = light_pdf(scene, t * t, dot(geometric_normals, wo))
455            radiance[pixel] += emitted * mis_weight(previous_pdf, pdf_light)[:, None]
456
457        # 光源は光を反射しないものとし、光源に当たったパスはここで終了する
458        alive = ids != scene.light_index
459        pixel, throughput, ids = pixel[alive], throughput[alive], ids[alive]
460        points, normals, wo = points[alive], normals[alive], wo[alive]
461
462        if use_nee:
463            radiance[pixel] += throughput * direct_lighting(scene, table, ids, points, normals, wo, rng)
464
465        new_directions, weight, previous_pdf = sample_brdf(table, ids, normals, wo, rng)
466        throughput = throughput * weight
467
468        # ロシアンルーレット：重みの小さいパスほど高い確率で打ち切り、生き残ったパスの重みを増やす
469        survive = throughput.max(axis=1) > 0.0
470        if depth >= 3:
471            q = np.clip(throughput.max(axis=1), 0.05, 0.95)
472            survive &= rng.random(len(q)) < q
473            throughput = throughput / q[:, None]
474
475        pixel, throughput, previous_pdf = pixel[survive], throughput[survive], previous_pdf[survive]
476        origins = points[survive] + normals[survive] * EPSILON
477        directions = new_directions[survive]
478
479    return radiance

482def generate_camera_rays(camera: Camera, width: int, height: int,
483                         rng: np.random.Generator) -> tuple[FloatArray, FloatArray]:
484    """画素ごとに 1 本ずつカメラレイを生成する。画素内の位置はランダムにずらす。"""
485    position = np.array(camera.position)
486    forward = np.array(camera.look_at) - position
487    forward /= np.linalg.norm(forward)
488    right = np.cross(forward, np.array(camera.up))
489    right /= np.linalg.norm(right)
490    up = np.cross(right, forward)
491
492    rows, cols = np.mgrid[0:height, 0:width]
493    rows, cols = rows.ravel(), cols.ravel()
494    jitter = rng.random((len(rows), 2))
495    tan_half = math.tan(math.radians(camera.fov_degrees) / 2.0)
496    x = (2.0 * (cols + jitter[:, 0]) / width - 1.0) * tan_half * width / height
497    y = (1.0 - 2.0 * (rows + jitter[:, 1]) / height) * tan_half
498
499    directions = normalize(forward + x[:, None] * right + y[:, None] * up)
500    origins = np.tile(position, (len(rows), 1))
501    return origins, directions

504def render(scene: Scene, width: int, height: int, spp: int, max_depth: int = 8,
505           use_nee: bool = True, seed: int = 0) -> FloatArray:
506    """シーンを描画し、線形な放射輝度の画像（高さ × 幅 × 3）を返す。"""
507    rng = np.random.default_rng(seed)
508    table = build_material_table(scene)
509    accumulated = np.zeros((width * height, 3))
510    for _ in range(spp):
511        origins, directions = generate_camera_rays(scene.camera, width, height, rng)
512        accumulated += trace_paths(scene, table, origins, directions, max_depth, use_nee, rng)
513    result: FloatArray = (accumulated / spp).reshape(height, width, 3)

32def fresnel_reflectance(cos_i: FloatArray, eta: complex) -> FloatArray:
33    """空気（屈折率 1）から複素屈折率 eta の物質へ入射する光の反射率（偏光なし）を返す。
34
35    s 偏光と p 偏光の反射率の平均をとる。eta の虚部を 0 にすれば誘電体、
36    虚部が大きければ金属を表す。
37    """
38    cos_i_c = cos_i.astype(np.complex128)
39    sin2_t = (1.0 - cos_i_c ** 2) / eta ** 2      # スネルの法則
40    cos_t = np.sqrt(1.0 - sin2_t)
41    r_s = (cos_i_c - eta * cos_t) / (cos_i_c + eta * cos_t)
42    r_p = (eta * cos_i_c - cos_t) / (eta * cos_i_c + cos_t)
43    result: FloatArray = 0.5 * (np.abs(r_s) ** 2 + np.abs(r_p) ** 2)
44    return result

47def schlick(cos_i: FloatArray, f0: float) -> FloatArray:
48    """Schlick 近似 F0 + (1 - F0)(1 - cosθ)^5 を返す。"""
49    result: FloatArray = f0 + (1.0 - f0) * (1.0 - cos_i) ** 5
50    return result

 30def ggx_distribution(cos_h: FloatArray, alpha: float) -> FloatArray:
 31    """GGX（Trowbridge-Reitz）法線分布関数 D。"""
 32    a2 = alpha * alpha
 33    denom = cos_h * cos_h * (a2 - 1.0) + 1.0
 34    result: FloatArray = a2 / (math.pi * denom * denom)
 35    return result

 38def smith_g1(cos_v: FloatArray, alpha: float) -> FloatArray:
 39    """GGX に対する Smith のマスキング関数 G1。"""
 40    a2 = alpha * alpha
 41    result: FloatArray = 2.0 * cos_v / (cos_v + np.sqrt(a2 + (1.0 - a2) * cos_v * cos_v))
 42    return result

 45def fresnel_schlick(cos_theta: FloatArray, f0: FloatArray) -> FloatArray:
 46    """フレネル反射率の Schlick 近似。"""
 47    result: FloatArray = f0 + (1.0 - f0) * (1.0 - cos_theta) ** 5
 48    return result

 51def cook_torrance_specular(cos_i: FloatArray, cos_o: FloatArray, cos_h: FloatArray,
 52                           alpha: float) -> FloatArray:
 53    """フレネル項を 1 とした Cook-Torrance 鏡面反射項 D G / (4 cosθ_i cosθ_o)。"""
 54    d = ggx_distribution(cos_h, alpha)
 55    g = smith_g1(cos_i, alpha) * smith_g1(cos_o, alpha)
 56    result: FloatArray = d * g / (4.0 * cos_i * cos_o)
 57    return result

119def shade_sphere(base_color: tuple[float, float, float], metallic: float, roughness: float,
120                 size: int = 128) -> FloatArray:
121    """平行光源 1 つで照らした球を、メタリック／ラフネスモデルでシェーディングする。
122
123    戻り値は線形な放射輝度の画像（size × size × 3）。背景は暗い灰色とする。
124    """
125    rows, cols = np.mgrid[0:size, 0:size]
126    x = (cols + 0.5) / size * 2.0 - 1.0
127    y = 1.0 - (rows + 0.5) / size * 2.0
128    r2 = x * x + y * y
129    inside = r2 < 1.0
130    z = np.sqrt(np.maximum(0.0, 1.0 - r2))
131    normals = np.stack([x, y, z], axis=-1)
132
133    view = np.array([0.0, 0.0, 1.0])
134    light = np.array([-0.5, 0.6, 0.62])
135    light /= np.linalg.norm(light)
136    half = (view + light) / np.linalg.norm(view + light)
137    light_irradiance = 3.0                     # 光に垂直な面での放射照度
138    alpha = max(roughness ** 2, 1e-3)
139
140    cos_i = np.maximum(normals @ light, 0.0)
141    cos_o = np.maximum(normals @ view, 1e-4)
142    cos_h = np.maximum(normals @ half, 0.0)
143    cos_oh = float(view @ half)
144
145    base = np.array(base_color)
146    f0 = 0.04 * (1.0 - metallic) + base * metallic       # 非金属は F0 = 0.04
147    fresnel = fresnel_schlick(np.full(3, cos_oh), f0)
148    d = ggx_distribution(cos_h, alpha)
149    g = smith_g1(np.maximum(cos_i, 1e-4), alpha) * smith_g1(cos_o, alpha)
150    specular = fresnel * (d * g / (4.0 * np.maximum(cos_i, 1e-4) * cos_o))[..., None]
151    diffuse = (1.0 - fresnel) * (1.0 - metallic) * base / math.pi
152    ambient = 0.03 * base                                 # 影側が真っ黒にならないための環境光
153
154    color = (diffuse + specular) * (light_irradiance * cos_i)[..., None] + ambient
155    background = np.full(3, 0.02)
156    result: FloatArray = np.where(inside[..., None], color, background)
157    return result

  1"""モンテカルロ積分の収束と重点サンプリングの効果を確かめる。
  2
  31. 正方形内に一様に点を打ち、四分円に入った割合から円周率を推定する。
  42. 半球上の積分 ∫ cos^3θ dω（厳密値は π / 2）を、一様サンプリングと
  5   コサイン重み付きサンプリングの 2 通りで推定し、誤差を比べる。
  6
  7実行例::
  8
  9    python monte_carlo.py --output-dir .
 10"""
 11
 12import argparse
 13import math
 14from collections.abc import Callable
 15from pathlib import Path
 16
 17import matplotlib
 18import numpy as np
 19import numpy.typing as npt
 20
 21matplotlib.use("Agg")
 22import matplotlib.pyplot as plt  # noqa: E402
 23
 24FloatArray = npt.NDArray[np.float64]
 25Estimator = Callable[[int, np.random.Generator], float]
 26
 27
 28def estimate_pi(n: int, rng: np.random.Generator) -> float:
 29    """n 個の乱数点を使って円周率を推定する。"""
 30    x = rng.random(n)
 31    y = rng.random(n)
 32    inside = x * x + y * y <= 1.0
 33    return 4.0 * float(np.mean(inside))
 34
 35
 36def integrand(cos_theta: FloatArray) -> FloatArray:
 37    """被積分関数 cos^3θ。半球上で積分すると π / 2 になる。"""
 38    result: FloatArray = cos_theta ** 3
 39    return result
 40
 41
 42EXACT_VALUE = math.pi / 2.0
 43
 44
 45def estimate_uniform(n: int, rng: np.random.Generator) -> float:
 46    """半球上の一様サンプリング（pdf = 1 / 2π）による推定値。
 47
 48    半球上の一様な方向では cosθ が [0, 1] の一様分布に従う。
 49    被積分関数が θ だけに依存するので、cosθ だけをサンプリングすればよい。
 50    """
 51    cos_theta = rng.random(n)
 52    pdf = 1.0 / (2.0 * math.pi)
 53    return float(np.mean(integrand(cos_theta) / pdf))
 54
 55
 56def estimate_cosine(n: int, rng: np.random.Generator) -> float:
 57    """コサイン重み付きサンプリング（pdf = cosθ / π）による推定値。"""
 58    cos_theta = np.sqrt(1.0 - rng.random(n))
 59    pdf = cos_theta / math.pi
 60    return float(np.mean(integrand(cos_theta) / pdf))
 61
 62
 63def rms_error(estimator: Estimator, exact: float, n: int, trials: int, rng: np.random.Generator) -> float:
 64    """推定を trials 回繰り返したときの二乗平均平方根誤差を返す。"""
 65    errors = np.array([estimator(n, rng) - exact for _ in range(trials)])
 66    return float(np.sqrt(np.mean(errors ** 2)))
 67
 68
 69def sample_counts() -> list[int]:
 70    """グラフの横軸に使うサンプル数（10 から 100000 まで対数的に等間隔）。"""
 71    return sorted({int(round(n)) for n in np.logspace(1, 5, 17)})

 74def plot_convergence(output: Path, seed: int = 0) -> None:
 75    """円周率推定のサンプル数と誤差の関係を両対数グラフで描く。"""
 76    rng = np.random.default_rng(seed)
 77    counts = sample_counts()
 78    errors = [rms_error(estimate_pi, math.pi, n, 100, rng) for n in counts]
 79    reference = errors[0] * math.sqrt(counts[0]) / np.sqrt(counts)
 80
 81    fig, ax = plt.subplots(figsize=(6.4, 3.8))
 82    ax.loglog(counts, errors, "o-", label="RMS error of the estimate")
 83    ax.loglog(counts, reference, "--", color="gray", label=r"$\propto 1/\sqrt{N}$")
 84    ax.set_xlabel("Number of samples N")
 85    ax.set_ylabel("RMS error")
 86    ax.grid(alpha=0.3, which="both")
 87    ax.legend()
 88    fig.tight_layout()
 89    fig.savefig(output, dpi=120)
 90    plt.close(fig)
 91
 92
 93def plot_importance(output: Path, seed: int = 0) -> None:
 94    """一様サンプリングとコサイン重み付きサンプリングの誤差を比べる。"""
 95    rng = np.random.default_rng(seed)
 96    counts = sample_counts()
 97    fig, ax = plt.subplots(figsize=(6.4, 3.8))
 98    for name, estimator in [("Uniform hemisphere sampling", estimate_uniform),
 99                            ("Cosine-weighted sampling", estimate_cosine)]:
100        errors = [rms_error(estimator, EXACT_VALUE, n, 100, rng) for n in counts]
101        ax.loglog(counts, errors, "o-", label=name)
102    ax.set_xlabel("Number of samples N")
103    ax.set_ylabel("RMS error")
104    ax.grid(alpha=0.3, which="both")
105    ax.legend()
106    fig.tight_layout()
107    fig.savefig(output, dpi=120)
108    plt.close(fig)

112    parser = argparse.ArgumentParser(description="モンテカルロ積分の実験")
113    parser.add_argument("--output-dir", type=Path, default=Path("."), help="出力先フォルダ")
114    args = parser.parse_args()
115
116    rng = np.random.default_rng(0)
117    for n in [100, 10_000, 1_000_000]:
118        print(f"N = {n:>9}: 円周率の推定値 = {estimate_pi(n, rng):.5f}")
119    for n in [16, 256]:
120        print(f"N = {n:>4}: 一様 = {estimate_uniform(n, rng):.4f}, "
121              f"コサイン重み付き = {estimate_cosine(n, rng):.4f}, 厳密値 = {EXACT_VALUE:.4f}")
122
123    args.output_dir.mkdir(parents=True, exist_ok=True)
124    plot_convergence(args.output_dir / "mc_convergence.png")

 29def srgb_encode(linear: FloatArray) -> FloatArray:
 30    """線形な値（0〜1）を sRGB の符号化値（0〜1）に変換する。"""
 31    c = np.clip(linear, 0.0, 1.0)
 32    result: FloatArray = np.where(c <= 0.0031308, 12.92 * c, 1.055 * c ** (1.0 / 2.4) - 0.055)
 33    return result
 34
 35
 36def srgb_decode(encoded: FloatArray) -> FloatArray:
 37    """sRGB の符号化値（0〜1）を線形な値（0〜1）に戻す。"""
 38    c = np.clip(encoded, 0.0, 1.0)
 39    result: FloatArray = np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)
 40    return result
 41
 42
 43def clamp(x: FloatArray) -> FloatArray:
 44    """1 を超える値を 1 に切り詰めるだけの変換。"""
 45    result: FloatArray = np.clip(x, 0.0, 1.0)
 46    return result
 47
 48
 49def reinhard(x: FloatArray) -> FloatArray:
 50    """Reinhard の単純な演算子 x / (1 + x)。"""
 51    result: FloatArray = x / (1.0 + x)
 52    return result
 53
 54
 55def aces_approx(x: FloatArray) -> FloatArray:
 56    """Narkowicz による ACES フィルム調カーブの近似式。"""
 57    a, b, c, d, e = 2.51, 0.03, 2.43, 0.59, 0.14
 58    result: FloatArray = np.clip((x * (a * x + b)) / (x * (c * x + d) + e), 0.0, 1.0)
 59    return result

104def make_hdr_ramp(width: int = 512, height: int = 40) -> FloatArray:
105    """左端 2^-4 から右端 2^6 まで、明るさが指数的に増える HDR 画像を作る。
106
107    上半分は無彩色、下半分はオレンジ色とする。
108    """
109    exposure = 2.0 ** np.linspace(-4.0, 6.0, width)
110    gray = np.ones(3)
111    orange = np.array([1.0, 0.45, 0.1])
112    top = exposure[None, :, None] * gray
113    bottom = exposure[None, :, None] * orange
114    result: FloatArray = np.concatenate([np.repeat(top, height // 2, axis=0),
115                                         np.repeat(bottom, height // 2, axis=0)], axis=0)
116    return result
