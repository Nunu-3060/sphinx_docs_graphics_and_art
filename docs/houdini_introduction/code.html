本資料で使用した examples ディレクトリのファイルをまとめたページである。各章の本文では、ファイルの一部だけを示していることがある。全体を通して読みたい場合は、このページを使う。

examples.zip から全てのファイルをまとめてダウンロードできる。展開すると examples ディレクトリができ、その中に Python のスクリプト 14 個、vex ディレクトリに VEX のコード 7 個、静的検査の設定ファイル setup.cfg が入っている。

スクリプトの実行方法は、はじめに の「サンプルコードの使い方」で説明している。全ての Python のスクリプトは、flake8（1 行 120 文字まで）と mypy（strict = True）の検査で違反が無いことを確かめている。

Python のスクリプト

common.py

全てのスクリプトで使う共通の関数を定義したモジュールである。同じ名前のノードを作り直す関数、VEX のファイルを読む関数、エラーを確かめる関数、hip ファイルを保存する関数がある。単独では実行しない。

examples/common.py

 1"""サンプルスクリプトで共通に使う関数。
 2
 3Houdini の Python Shell や hython から使う。単独では実行しない。
 4"""
 5
 6from __future__ import annotations
 7
 8import sys
 9from pathlib import Path
10
11import hou
12
13# このファイルがあるディレクトリ（examples）。VEX のファイルを探すときに使う。
14EXAMPLES_DIR = Path(__file__).resolve().parent
15
16
17def read_vex(name: str) -> str:
18    """examples/vex/ にある VEX のコードを読み込んで返す。"""
19    return (EXAMPLES_DIR / "vex" / name).read_text(encoding="utf-8")
20
21
22def recreate_node(parent_path: str, node_type: str, name: str) -> hou.Node:
23    """parent_path の下に node_type のノードを name という名前で作る。
24
25    同じ名前のノードが既にあれば削除してから作る。スクリプトを何度実行しても
26    同じ結果になるようにするためである。
27    """
28    parent = hou.node(parent_path)
29    old = parent.node(name)
30    if old is not None:
31        old.destroy()
32    return parent.createNode(node_type, name)
33
34
35def recreate_geo(name: str) -> hou.Node:
36    """/obj の下に Geometry オブジェクトを作り直して返す。"""
37    return recreate_node("/obj", "geo", name)
38
39
40def finish(network: hou.Node, output: hou.Node) -> None:
41    """output にディスプレイフラグとレンダーフラグを立て、ノードを整列する。"""
42    output.setDisplayFlag(True)
43    output.setRenderFlag(True)
44    network.layoutChildren()
45
46
47def check_errors(network: hou.Node) -> None:
48    """network の直下の全ノードをクックし、エラーがあれば例外を投げる。"""
49    for node in network.children():
50        try:
51            node.cook(force=True)
52        except hou.OperationFailed:
53            pass    # エラーの内容は node.errors() で調べる
54        errors = node.errors()
55        if errors:
56            raise RuntimeError(f"{node.path()}: {errors}")
57
58
59def save_hip_from_argv() -> None:
60    """コマンドライン引数でファイル名が渡されていれば、シーンを保存する。
61
62    hython で実行したときに使う。例: hython build_table.py table.hipnc
63    """
64    if len(sys.argv) > 1:
65        hou.hipFile.save(sys.argv[1])
66        print(f"保存した: {sys.argv[1]}")

build_table.py

はじめてのモデリング の机を作る。

examples/build_table.py

  1r"""「はじめてのモデリング」の机を、ノードを組み立てて作る。
  2
  3Houdini の Python Shell で次のように実行する（パスは各自の環境に合わせる）。
  4
  5    import sys
  6    sys.path.append(r"C:\houdini_intro\examples")
  7    import build_table
  8    build_table.build()
  9
 10hython からは ``hython build_table.py table.hipnc`` のように実行する。
 11"""
 12
 13from __future__ import annotations
 14
 15import hou
 16
 17from common import check_errors, finish, recreate_geo, save_hip_from_argv
 18
 19# 机の寸法（単位はメートル）。
 20TOP_SIZE = (1.2, 0.04, 0.7)   # 天板の幅、厚さ、奥行き
 21TABLE_HEIGHT = 0.72           # 床から天板の上面までの高さ
 22LEG_SIZE = 0.05               # 脚の断面（正方形）の一辺
 23LEG_INSET = 0.08              # 天板の端から脚の中心までの距離
 24HOLE_RADIUS = 0.03            # 配線用の穴の半径
 25
 26
 27def build() -> hou.Node:
 28    """/obj/table に机のネットワークを作り、出力の Null ノードを返す。"""
 29    geo = recreate_geo("table")
 30    top_w, top_t, top_d = TOP_SIZE
 31    leg_h = TABLE_HEIGHT - top_t
 32
 33    # 天板: 箱を作り、脚の上に載る高さまで持ち上げる。
 34    top = geo.createNode("box", "top")
 35    top.parmTuple("size").set(TOP_SIZE)
 36    lift = geo.createNode("xform", "lift_top")
 37    lift.setFirstInput(top)
 38    lift.parm("ty").set(leg_h + top_t / 2)
 39
 40    # 天板の角を丸める。
 41    bevel = geo.createNode("polybevel::3.0", "bevel_top")
 42    bevel.setFirstInput(lift)
 43    bevel.parm("offset").set(0.005)
 44
 45    # 配線用の穴: 円柱を天板の奥側に置き、天板から引く。
 46    hole = geo.createNode("tube", "hole")
 47    hole.parm("type").set("poly")
 48    hole.parm("cap").set(True)
 49    hole.parm("rad1").set(HOLE_RADIUS)
 50    hole.parm("rad2").set(HOLE_RADIUS)
 51    hole.parm("height").set(0.2)
 52    hole.parm("cols").set(24)
 53    hole.parmTuple("t").set((top_w / 2 - 0.15, leg_h, -(top_d / 2 - 0.1)))
 54    boolean = geo.createNode("boolean::2.0", "cut_hole")
 55    boolean.setInput(0, bevel)
 56    boolean.setInput(1, hole)
 57    boolean.parm("booleanop").set("subtract")
 58
 59    # 脚: 底面が y = 0 に来るように中心を高さの半分だけ上げた箱。
 60    leg = geo.createNode("box", "leg")
 61    leg.parmTuple("size").set((LEG_SIZE, leg_h, LEG_SIZE))
 62    leg.parm("ty").set(leg_h / 2)
 63
 64    # 脚を置く位置: 2 x 2 の点を持つグリッド（XZ 平面）。
 65    positions = geo.createNode("grid", "leg_positions")
 66    positions.parm("orient").set("zx")
 67    positions.parm("sizex").set(top_w - 2 * LEG_INSET)
 68    positions.parm("sizey").set(top_d - 2 * LEG_INSET)
 69    positions.parm("rows").set(2)
 70    positions.parm("cols").set(2)
 71
 72    # 脚をグリッドの 4 つの点にコピーする。グリッドの面の向き（上向き）に
 73    # 脚の Z 軸が合わせられて脚が倒れないように、Use Implicit N を切る。
 74    copy = geo.createNode("copytopoints::2.0", "copy_legs")
 75    copy.setInput(0, leg)
 76    copy.setInput(1, positions)
 77    copy.parm("useimplicitn").set(False)
 78
 79    # 天板と脚をまとめる。
 80    merge = geo.createNode("merge", "merge")
 81    merge.setInput(0, boolean)
 82    merge.setInput(1, copy)
 83
 84    out = geo.createNode("null", "OUT")
 85    out.setFirstInput(merge)
 86    finish(geo, out)
 87    return out
 88
 89
 90def main() -> None:
 91    out = build()
 92    check_errors(out.parent())
 93    geometry = out.geometry()
 94    print(f"ポイント数: {len(geometry.points())}")
 95    print(f"プリミティブ数: {len(geometry.prims())}")
 96    print(f"バウンディングボックス: {geometry.boundingBox()}")
 97    save_hip_from_argv()
 98
 99
100if __name__ == "__main__":
101    main()

build_attributes.py

ジオメトリとアトリビュート の、起伏のある地形に色を付ける例を作る。

examples/build_attributes.py

  1r"""「ジオメトリとアトリビュート」の例を作る。
  2
  3グリッドをノイズで起伏させて地形にし、法線（N）、高さ（height）、高さに応じた
  4色（Cd）、ランダムな大きさ（pscale）、高い所のポイントのグループ（peak）を付ける。
  5
  6Python Shell での実行例:
  7
  8    import sys
  9    sys.path.append(r"C:\houdini_intro\examples")
 10    import build_attributes
 11    build_attributes.build()
 12"""
 13
 14from __future__ import annotations
 15
 16import hou
 17
 18from common import check_errors, finish, recreate_geo, save_hip_from_argv
 19
 20
 21def build() -> hou.Node:
 22    """/obj/terrain にアトリビュートの例のネットワークを作り、出力の Null ノードを返す。"""
 23    geo = recreate_geo("terrain")
 24
 25    grid = geo.createNode("grid", "grid")
 26    grid.parm("rows").set(50)
 27    grid.parm("cols").set(50)
 28
 29    # ポイントごとの法線（N）を付ける。平らなグリッドなので全て真上を向く。
 30    normal = geo.createNode("normal", "normal")
 31    normal.setFirstInput(grid)
 32    normal.parm("type").set("typepoint")
 33
 34    # 位置（P）にノイズを加える。Noise Along Vector をオンにすると、
 35    # ノイズは法線（N）の方向、つまり真上にだけ加わる。
 36    noise = geo.createNode("attribnoise::2.0", "bumps")
 37    noise.setFirstInput(normal)
 38    noise.parm("attribtype").set("vec")
 39    noise.parm("attribs").set("P")
 40    noise.parm("displace").set(True)
 41    noise.parm("displaceattrib").set("N")
 42    noise.parm("amplitude").set(1.5)
 43    noise.parm("elementsize").set(3.0)
 44
 45    # 起伏を付けた後の法線を計算し直す。
 46    renormal = geo.createNode("normal", "recompute_normal")
 47    renormal.setFirstInput(noise)
 48    renormal.parm("type").set("typepoint")
 49
 50    # 各ポイントの高さ（P の y 成分）を height アトリビュートにコピーする。
 51    # $TY は「処理中のポイントの y 座標」を表すローカル変数である。
 52    height = geo.createNode("attribcreate::2.0", "height")
 53    height.setFirstInput(renormal)
 54    height.parm("name1").set("height")
 55    height.parm("value1v1").setExpression("$TY")
 56
 57    # height の値に応じて色（Cd）を付ける。低い所は緑、高い所は白にする。
 58    color = geo.createNode("color", "color_by_height")
 59    color.setFirstInput(height)
 60    color.parm("colortype").set("3")       # Ramp from Attribute
 61    color.parm("rampattribute").set("height")
 62    color.parm("ramprange1").set(0.0)
 63    color.parm("ramprange2").set(1.5)
 64    color.parm("ramp").set(hou.Ramp(
 65        (hou.rampBasis.Linear, hou.rampBasis.Linear),
 66        (0.0, 1.0),
 67        ((0.15, 0.35, 0.1), (0.95, 0.95, 0.95)),
 68    ))
 69
 70    # ポイントごとにランダムな大きさ（pscale）を付ける。
 71    randomize = geo.createNode("attribrandomize", "random_pscale")
 72    randomize.setFirstInput(color)
 73    randomize.parm("name").set("pscale")
 74    randomize.parm("dimensions").set(1)
 75    randomize.parm("minx").set(0.05)
 76    randomize.parm("maxx").set(0.15)
 77
 78    # height が 1 以上のポイントを peak グループに入れる。
 79    group = geo.createNode("groupexpression", "peak_group")
 80    group.setFirstInput(randomize)
 81    group.parm("grouptype").set("point")
 82    group.parm("groupname1").set("peak")
 83    group.parm("snippet1").set("@height >= 1.0")
 84
 85    out = geo.createNode("null", "OUT")
 86    out.setFirstInput(group)
 87    finish(geo, out)
 88    return out
 89
 90
 91def main() -> None:
 92    out = build()
 93    check_errors(out.parent())
 94    geometry = out.geometry()
 95    print("ポイントアトリビュート:", [a.name() for a in geometry.pointAttribs()])
 96    print("ポイントグループ:", [g.name() for g in geometry.pointGroups()])
 97    print(f"peak グループのポイント数: {len(geometry.findPointGroup('peak').points())}")
 98    save_hip_from_argv()
 99
100
101if __name__ == "__main__":
102    main()

build_expressions.py

パラメーターとエクスプレッション の、弾みながら回るボールを作る。

examples/build_expressions.py

 1r"""「パラメーターとエクスプレッション」の例を作る。
 2
 3- 床の大きさを、ボールの半径から ch() で計算する。
 4- ボールの高さを、フレーム番号 $F を使ったエクスプレッションで弾ませる。
 5- ボール全体を、キーフレームで 1 回転させる。
 6
 7Python Shell での実行例:
 8
 9    import sys
10    sys.path.append(r"C:\houdini_intro\examples")
11    import build_expressions
12    build_expressions.build()
13"""
14
15from __future__ import annotations
16
17import hou
18
19from common import check_errors, finish, recreate_geo, save_hip_from_argv
20
21
22def build() -> hou.Node:
23    """/obj/expressions に例のネットワークを作り、出力の Null ノードを返す。"""
24    geo = recreate_geo("expressions")
25
26    ball = geo.createNode("sphere", "ball")
27    ball.parm("type").set("polymesh")
28    ball.parmTuple("rad").set((0.5, 0.5, 0.5))
29
30    # 高さ: |sin($F * 6)| は 30 フレームごとに 0 に戻る（Hscript の sin は度で計算する）。
31    bounce = geo.createNode("xform", "bounce")
32    bounce.setFirstInput(ball)
33    bounce.parm("ty").setExpression('ch("../ball/radx") + abs(sin($F * 6)) * 2')
34
35    # 床: 一辺をボールの半径の 8 倍にする。
36    floor = geo.createNode("grid", "floor")
37    floor.parm("sizex").setExpression('ch("../ball/radx") * 8')
38    floor.parm("sizey").setExpression('ch("sizex")')
39
40    merge = geo.createNode("merge", "merge")
41    merge.setInput(0, bounce)
42    merge.setInput(1, floor)
43
44    # 回転: 1 フレーム目に 0 度、120 フレーム目に 360 度のキーフレームを打つ。
45    spin = geo.createNode("xform", "spin")
46    spin.setFirstInput(merge)
47    spin.parm("tx").set(1.0)
48    for frame, angle in ((1, 0.0), (120, 360.0)):
49        key = hou.Keyframe()
50        key.setFrame(frame)
51        key.setValue(angle)
52        key.setExpression("linear()")
53        spin.parm("ry").setKeyframe(key)
54
55    out = geo.createNode("null", "OUT")
56    out.setFirstInput(spin)
57    finish(geo, out)
58    return out
59
60
61def main() -> None:
62    out = build()
63    check_errors(out.parent())
64    bounce = hou.node("/obj/expressions/bounce")
65    spin = hou.node("/obj/expressions/spin")
66    for frame in (1, 15, 30, 60, 120):
67        hou.setFrame(frame)
68        print(f"フレーム {frame:3d}: 高さ {bounce.parm('ty').eval():.3f}, 回転 {spin.parm('ry').eval():.1f} 度")
69    save_hip_from_argv()
70
71
72if __name__ == "__main__":
73    main()

build_vex_examples.py

VEX 入門 の 4 つの例を作る。VEX のコードは vex ディレクトリのファイルから読み込む。

examples/build_vex_examples.py

 1r"""「VEX 入門」の例を作る。
 2
 3/obj/vex_examples の中に、examples/vex/ の 4 つの VEX のコードを
 4それぞれ Attribute Wrangle に設定したネットワークを作る。
 5
 6Python Shell での実行例:
 7
 8    import sys
 9    sys.path.append(r"C:\houdini_intro\examples")
10    import build_vex_examples
11    build_vex_examples.build()
12"""
13
14from __future__ import annotations
15
16import hou
17
18from common import check_errors, finish, read_vex, recreate_geo, save_hip_from_argv
19
20
21def add_wrangle(geo: hou.Node, name: str, run_over: str, vex_file: str,
22                parms: dict[str, float]) -> hou.Node:
23    """VEX のコードとスペアパラメーターを設定した Attribute Wrangle を作る。
24
25    parms には「パラメーター名: 初期値」を渡す。初期値が int なら整数の、
26    float なら浮動小数点数のパラメーターを作る。
27    """
28    wrangle = geo.createNode("attribwrangle", name)
29    wrangle.parm("class").set(run_over)
30    wrangle.parm("snippet").set(read_vex(vex_file))
31    for parm_name, value in parms.items():
32        template: hou.ParmTemplate
33        if isinstance(value, int):
34            template = hou.IntParmTemplate(parm_name, parm_name, 1, default_value=(value,))
35        else:
36            template = hou.FloatParmTemplate(parm_name, parm_name, 1, default_value=(value,))
37        wrangle.addSpareParmTuple(template)
38    return wrangle
39
40
41def build() -> hou.Node:
42    """/obj/vex_examples に例のネットワークを作り、波の例の出力ノードを返す。"""
43    geo = recreate_geo("vex_examples")
44
45    # 例 1: 波打つグリッド。
46    grid = geo.createNode("grid", "grid")
47    grid.parm("rows").set(60)
48    grid.parm("cols").set(60)
49    wave = add_wrangle(geo, "wave", "point", "wave.vex", {"frequency": 2.0, "amplitude": 0.3})
50    wave.setFirstInput(grid)
51    wave_out = geo.createNode("null", "OUT_wave")
52    wave_out.setFirstInput(wave)
53
54    # 例 2: でこぼこの球。法線 N を持つ球を作るため、Normal SOP を通す。
55    sphere = geo.createNode("sphere", "sphere")
56    sphere.parm("type").set("polymesh")
57    sphere.parm("rows").set(60)
58    sphere.parm("cols").set(120)
59    normal = geo.createNode("normal", "normal")
60    normal.setFirstInput(sphere)
61    normal.parm("type").set("typepoint")
62    bumpy = add_wrangle(geo, "noise_displace", "point", "noise_displace.vex",
63                        {"frequency": 3.0, "amplitude": 0.4})
64    bumpy.setFirstInput(normal)
65    bumpy_out = geo.createNode("null", "OUT_noise")
66    bumpy_out.setFirstInput(bumpy)
67
68    # 例 3: らせん。入力は無い。
69    spiral = add_wrangle(geo, "spiral", "detail", "spiral.vex", {"count": 200, "turns": 5.0, "height": 3.0})
70    spiral_out = geo.createNode("null", "OUT_spiral")
71    spiral_out.setFirstInput(spiral)
72
73    # 例 4: らせんに最も近い距離で、グリッドに色を付ける。
74    nearest = add_wrangle(geo, "nearest_color", "point", "nearest_color.vex", {"max_distance": 2.0})
75    nearest.setInput(0, grid)
76    nearest.setInput(1, spiral)
77    nearest_out = geo.createNode("null", "OUT_nearest")
78    nearest_out.setFirstInput(nearest)
79
80    finish(geo, wave_out)
81    return wave_out
82
83
84def main() -> None:
85    out = build()
86    check_errors(out.parent())
87    for name in ("OUT_wave", "OUT_noise", "OUT_spiral", "OUT_nearest"):
88        geometry = out.parent().node(name).geometry()
89        print(f"{name}: ポイント {len(geometry.points())}, プリミティブ {len(geometry.prims())}, "
90              f"範囲 {geometry.boundingBox()}")
91    save_hip_from_argv()
92
93
94if __name__ == "__main__":
95    main()

examples/build_city.py

  1r"""「プロシージャルモデリングの実践」の建物群を作る。
  2
  3グリッドの各マスを区画とし、区画ごとに乱数で決めた高さまで押し出して
  4建物にする。区画ごとの押し出しには For-Each ループを使う。
  5
  6Python Shell での実行例:
  7
  8    import sys
  9    sys.path.append(r"C:\houdini_intro\examples")
 10    import build_city
 11    build_city.build()
 12"""
 13
 14from __future__ import annotations
 15
 16import hou
 17
 18from common import check_errors, finish, read_vex, recreate_geo, save_hip_from_argv
 19
 20# ループの中で、処理中の区画（プリミティブ 0 番）の height アトリビュートを読む式。
 21HEIGHT_EXPR = 'prim(opinputpath(".", 0), 0, "height", 0)'
 22FLOOR_HEIGHT = 3.0   # 1 フロアの高さ（メートル）
 23MAX_HEIGHT = 60.0    # 建物の高さの上限（メートル）
 24
 25
 26def add_float_parm(node: hou.Node, name: str, label: str, default: float) -> None:
 27    """node に浮動小数点数のスペアパラメーターを 1 個追加する。"""
 28    node.addSpareParmTuple(hou.FloatParmTemplate(name, label, 1, default_value=(default,)))
 29
 30
 31def build() -> hou.Node:
 32    """/obj/city に建物群のネットワークを作り、出力の Null ノードを返す。"""
 33    geo = recreate_geo("city")
 34
 35    # 地面: 100 m 四方を 10 x 10 のマスに分けたグリッド。
 36    ground = geo.createNode("grid", "ground")
 37    ground.parmTuple("size").set((100.0, 100.0))
 38    ground.parm("rows").set(11)
 39    ground.parm("cols").set(11)
 40
 41    # 区画: 各マスを内側に縮め、マスごとに独立した面にする。
 42    lots = geo.createNode("polyextrude::2.0", "make_lots")
 43    lots.setFirstInput(ground)
 44    lots.parm("splittype").set("elements")
 45    lots.parm("inset").set(1.5)
 46    lots.parm("outputside").set(False)
 47
 48    # 区画ごとに高さを決め、一部を空き地にする。
 49    attrs = geo.createNode("attribwrangle", "lot_attributes")
 50    attrs.setFirstInput(lots)
 51    attrs.parm("class").set("primitive")
 52    attrs.parm("snippet").set(read_vex("lot_attributes.vex"))
 53    add_float_parm(attrs, "seed", "Seed", 0.0)
 54    add_float_parm(attrs, "empty_ratio", "Empty Ratio", 0.2)
 55    add_float_parm(attrs, "min_height", "Min Height", 6.0)
 56    add_float_parm(attrs, "max_height", "Max Height", MAX_HEIGHT)
 57
 58    # For-Each ループ: 区画（プリミティブ）を 1 個ずつ取り出して押し出す。
 59    begin = geo.createNode("block_begin", "foreach_begin")
 60    end = geo.createNode("block_end", "foreach_end")
 61    begin.setFirstInput(attrs)
 62    begin.parm("method").set("piece")
 63    begin.parm("blockpath").set("../foreach_end")
 64    end.parm("itermethod").set("pieces")
 65    end.parm("method").set("merge")
 66    end.parm("class").set("primitive")
 67    end.parm("useattrib").set(False)
 68    end.parm("blockpath").set("../foreach_begin")
 69    end.parm("templatepath").set("../foreach_begin")
 70
 71    extrude = geo.createNode("polyextrude::2.0", "extrude_building")
 72    extrude.setFirstInput(begin)
 73    extrude.parm("dist").setExpression(HEIGHT_EXPR)
 74    extrude.parm("divs").setExpression(f"max(1, floor({HEIGHT_EXPR} / {FLOOR_HEIGHT}))")
 75    end.setFirstInput(extrude)
 76
 77    # 高さに応じて色を付ける。
 78    color = geo.createNode("attribwrangle", "building_color")
 79    color.setFirstInput(end)
 80    color.parm("class").set("point")
 81    color.parm("snippet").set(read_vex("building_color.vex"))
 82    add_float_parm(color, "max_height", "Max Height", MAX_HEIGHT)
 83
 84    # 地面と建物をまとめる。
 85    merge = geo.createNode("merge", "merge")
 86    merge.setInput(0, ground)
 87    merge.setInput(1, color)
 88
 89    out = geo.createNode("null", "OUT")
 90    out.setFirstInput(merge)
 91    finish(geo, out)
 92    return out
 93
 94
 95def main() -> None:
 96    out = build()
 97    check_errors(out.parent())
 98    geometry = out.geometry()
 99    print(f"プリミティブ数: {len(geometry.prims())}")
100    print(f"バウンディングボックス: {geometry.boundingBox()}")
101    save_hip_from_argv()
102
103
104if __name__ == "__main__":
105    main()

python_sop_wave.py

Python による操作 で使う。Python SOP の Python Code に貼り付けるコードである。

examples/python_sop_wave.py

 1"""Python SOP に貼り付けて使うコード。波打つ面をポイントとポリゴンから作る。
 2
 3build_python_sop.py が、このファイルの中身を Python SOP の Python Code
 4パラメーターに設定する。Python SOP の中では、hou.pwd() がその Python SOP
 5自身を表す。
 6"""
 7
 8from __future__ import annotations
 9
10import math
11
12import hou
13
14COLUMNS = 40      # x 方向のポイント数
15ROWS = 40         # z 方向のポイント数
16SIZE = 4.0        # 面の一辺の長さ
17AMPLITUDE = 0.2   # 波の高さ
18
19node = hou.pwd()
20geo = node.geometry()       # このノードが出力するジオメトリ（最初は空）
21phase = hou.frame() * 0.1   # フレームが進むと波が動く
22
23# ポイントを作る。i が z 方向、j が x 方向の番号である。
24positions = []
25for i in range(ROWS):
26    for j in range(COLUMNS):
27        x = (j / (COLUMNS - 1) - 0.5) * SIZE
28        z = (i / (ROWS - 1) - 0.5) * SIZE
29        y = AMPLITUDE * math.sin(3.0 * x + phase) * math.cos(2.0 * z)
30        positions.append(hou.Vector3(x, y, z))
31points = geo.createPoints(positions)
32
33# 隣り合う 4 つのポイントを結んで四角形のポリゴンを作る。Houdini では、
34# 表から見て時計回りにポイントを並べると、その面が表（法線の向き）になる。
35quads = []
36for i in range(ROWS - 1):
37    for j in range(COLUMNS - 1):
38        a = i * COLUMNS + j
39        quads.append((points[a], points[a + 1], points[a + COLUMNS + 1], points[a + COLUMNS]))
40geo.createPolygons(quads)

12def add_out_nulls() -> list[hou.Node]:
13    """選択中の SOP ノードそれぞれの後ろに OUT という Null ノードを作り、表示する。"""
14    created = []
15    for node in hou.selectedNodes():
16        if node.type().category() != hou.sopNodeTypeCategory():
17            continue    # SOP 以外のノードは対象外
18        null = node.parent().createNode("null", "OUT")   # 名前が重なると OUT1 などになる

21        null.setDisplayFlag(True)
22        null.setRenderFlag(True)
23        created.append(null)
24    if not created:
25        hou.ui.displayMessage("SOP ノードを選択してから実行する。")

examples/build_hda.py

  1r"""「デジタルアセット」の例: 階段のネットワークを HDA にする。
  2
  31. サブネットワークの中に階段のネットワークを作る。

 24
 25from common import check_errors, read_vex, recreate_geo, save_hip_from_argv
 26
 27# HDA のノードタイプ名。「名前空間::名前::バージョン」の形にする。
 28ASSET_NAME = "intro::stairs::1.0"
 29ASSET_LABEL = "Intro Stairs"
 30# HDA を保存するファイル。$HOUDINI_USER_PREF_DIR/otls の HDA は起動時に読み込まれる。
 31DEFAULT_HDA_PATH = "$HOUDINI_USER_PREF_DIR/otls/intro_stairs.hdanc"
 32
 33# HDA のパラメーター: (名前, ラベル, 初期値)。
 34PARMS: list[tuple[str, str, float]] = [
 35    ("step_count", "Step Count", 10),
 36    ("step_height", "Step Height", 0.18),
 37    ("step_depth", "Step Depth", 0.28),
 38    ("width", "Width", 1.0),
 39]
 40
 41
 42def build_contents(subnet: hou.Node) -> None:
 43    """サブネットワークの中に階段のネットワークを作る（build_stairs.py と同じ構成）。"""
 44    points = subnet.createNode("attribwrangle", "step_points")
 45    points.parm("class").set("detail")
 46    points.parm("snippet").set(read_vex("stairs_points.vex"))
 47
 48    box = subnet.createNode("box", "unit_box")
 49    copy = subnet.createNode("copytopoints::2.0", "copy_steps")
 50    copy.setInput(0, box)
 51    copy.setInput(1, points)
 52
 53    output = subnet.createNode("output", "output0")
 54    output.setFirstInput(copy)
 55    subnet.layoutChildren()
 56
 57
 58def setup_interface(asset: hou.Node) -> None:
 59    """HDA にパラメーターを追加し、中の Wrangle から参照させる。"""
 60    definition = asset.type().definition()
 61    group = definition.parmTemplateGroup()
 62    for name, label, default in PARMS:
 63        template: hou.ParmTemplate
 64        if isinstance(default, int):
 65            template = hou.IntParmTemplate(name, label, 1, default_value=(default,), min=1, max=50)
 66        else:
 67            template = hou.FloatParmTemplate(name, label, 1, default_value=(default,))
 68        group.append(template)
 69    definition.setParmTemplateGroup(group)
 70
 71    # VEX の chi("step_count") などは Wrangle 自身のパラメーターを読む。そこで
 72    # Wrangle にも同じ名前のパラメーターを作り、Hscript の ch() で HDA の値を参照させる。
 73    wrangle = asset.node("step_points")
 74    for name, label, default in PARMS:
 75        template = group.find(name)
 76        wrangle.addSpareParmTuple(template)
 77        wrangle.parm(name).setExpression(f'ch("../{name}")')
 78    definition.updateFromNode(asset)
 79
 80
 81def build(hda_path: str = DEFAULT_HDA_PATH) -> hou.Node:
 82    """/obj/hda_demo に階段の HDA を作り、そのノードを返す。"""
 83    geo = recreate_geo("hda_demo")
 84    subnet = geo.createNode("subnet", "stairs")
 85    build_contents(subnet)
 86
 87    file_path = hou.text.expandString(hda_path)
 88    asset = subnet.createDigitalAsset(
 89        name=ASSET_NAME,
 90        hda_file_name=file_path,
 91        description=ASSET_LABEL,
 92        min_num_inputs=0,
 93        max_num_inputs=0,
 94    )
 95    setup_interface(asset)
 96    asset.setDisplayFlag(True)
 97    asset.setRenderFlag(True)
 98    print(f"HDA を保存した: {file_path}")
 99    return asset
100
101
102def main() -> None:
103    # hython で試すときは、HDA を一時ディレクトリに保存する。
104    hda_path = str(Path(tempfile.gettempdir()) / "intro_stairs.hdanc")
105    asset = build(hda_path)
106    check_errors(asset.parent())
107    asset.parm("step_count").set(5)
108    print(f"5 段にしたときの範囲: {asset.geometry().boundingBox()}")
109
110    # 2 個目のインスタンスを作り、別の値を設定できることを確かめる。
111    second = asset.parent().createNode(ASSET_NAME, "stairs_wide")
112    second.parm("width").set(3.0)
113    print(f"2 個目（幅 3 m）の範囲: {second.geometry().boundingBox()}")
114    save_hip_from_argv()

examples/build_render.py

  1r"""「マテリアルとレンダリング」の例: 机を Solaris に読み込み、Karma でレンダリングする。
  2
  3/stage に次の LOP ネットワークを作る。
  4
  5    SOP Import（机） ─┐
  6    SOP Create（床） ─┴ Merge → Material Library → Camera → Dome Light
  7        → Distant Light → Karma Render Settings → USD Render ROP
  8
  9Python Shell での実行例（build_table.py で机を作ってから実行する）:
 10
 11    import sys
 12    sys.path.append(r"C:\houdini_intro\examples")
 13    import build_table, build_render
 14    build_table.build()
 15    rop = build_render.build()
 16    rop.render()    # $HIP/render/table.jpg に画像を書き出す
 17"""
 18
 19from __future__ import annotations
 20
 21import hou
 22
 23from common import check_errors, recreate_node, save_hip_from_argv
 24
 25TABLE_SOP = "/obj/table/OUT"
 26
 27
 28def add_material(library: hou.Node, name: str, color: tuple[float, float, float],
 29                 roughness: float, geometry: str) -> None:
 30    """Material Library の中に MtlX Standard Surface を作り、geometry に割り当てる。"""
 31    shader = library.createNode("mtlxstandard_surface", name)
 32    shader.parmTuple("base_color").set(color)
 33    shader.parm("specular_roughness").set(roughness)
 34
 35    # Material Library の「Materials」マルチパラメーターに 1 行追加する。
 36    index = library.parm("materials").eval() + 1
 37    library.parm("materials").set(index)
 38    library.parm(f"matnode{index}").set(name)
 39    library.parm(f"matpath{index}").set(f"/materials/{name}")
 40    library.parm(f"assign{index}").set(True)
 41    library.parm(f"geopath{index}").set(geometry)
 42
 43
 44def build(output_image: str = "$HIP/render/table.jpg") -> hou.Node:
 45    """/stage にレンダリング用の LOP ネットワークを作り、USD Render ROP を返す。"""
 46    stage = hou.node("/stage")
 47    for child in stage.children():
 48        child.destroy()
 49
 50    # 机: SOP のジオメトリを USD のプリミティブ /table として読み込む。
 51    table = recreate_node("/stage", "sopimport", "table")
 52    table.parm("soppath").set(TABLE_SOP)
 53    table.parm("primpath").set("/table")
 54

build_rbd.py

シミュレーション入門 の、箱の壁を崩す剛体シミュレーションを作る。

examples/build_rbd.py

 1r"""「シミュレーション入門」の例: 積んだ箱を球にぶつけて崩す剛体シミュレーション。
 2
 3RBD Bullet Solver SOP を使う。DOP ネットワークを自分で組まなくても、
 4SOP の中だけで剛体シミュレーションができる。
 5
 6Python Shell での実行例:
 7
 8    import sys
 9    sys.path.append(r"C:\houdini_intro\examples")
10    import build_rbd
11    build_rbd.build()
12
13作った後、プレイバーで再生するとシミュレーションが進む。
14"""
15
16from __future__ import annotations
17
18import hou
19
20from common import check_errors, finish, recreate_geo, save_hip_from_argv
21
22BOX_SIZE = 0.5       # 箱の一辺（メートル）
23COLUMNS = 6          # 横に並べる箱の数
24LAYERS = 6           # 積む段数
25
26
27def build() -> hou.Node:
28    """/obj/rbd に剛体シミュレーションのネットワークを作り、出力の Null ノードを返す。"""
29    geo = recreate_geo("rbd")
30
31    # 箱を積む位置: 横 COLUMNS 個 x 縦 LAYERS 段の格子状のポイント（XY 平面）。
32    positions = geo.createNode("grid", "wall_points")
33    positions.parm("orient").set("xy")
34    positions.parm("sizex").set(BOX_SIZE * (COLUMNS - 1))
35    positions.parm("sizey").set(BOX_SIZE * (LAYERS - 1))
36    positions.parm("rows").set(LAYERS)
37    positions.parm("cols").set(COLUMNS)
38    positions.parm("ty").set(BOX_SIZE * LAYERS / 2)
39
40    # 箱をコピーして壁にする。箱は少しだけ小さくし、初期状態で重ならないようにする。
41    box = geo.createNode("box", "brick")
42    box.parmTuple("size").set((BOX_SIZE * 0.98,) * 3)
43    wall = geo.createNode("copytopoints::2.0", "wall")
44    wall.setInput(0, box)
45    wall.setInput(1, positions)
46    wall.parm("useimplicitn").set(False)
47
48    # 壁に向かって飛んでいく球。初速度は v アトリビュートで与える。
49    ball = geo.createNode("sphere", "ball")
50    ball.parm("type").set("polymesh")
51    ball.parmTuple("rad").set((0.6, 0.6, 0.6))
52    ball.parmTuple("t").set((0.0, 1.2, 4.0))
53    velocity = geo.createNode("attribwrangle", "ball_velocity")
54    velocity.setFirstInput(ball)
55    velocity.parm("class").set("point")
56    velocity.parm("snippet").set("v@v = {0, 1, -12};   // 毎秒 12 m で -Z 方向へ")
57
58    # 箱と球を 1 つにまとめる。RBD Bullet Solver は、つながった部分（ここでは
59    # 箱 1 個、球 1 個）をそれぞれ 1 個の剛体として扱う。
60    merge = geo.createNode("merge", "merge")
61    merge.setInput(0, wall)
62    merge.setInput(1, velocity)
63
64    solver = geo.createNode("rbdbulletsolver", "rbd_solver")
65    solver.setFirstInput(merge)
66    solver.parm("useground").set(1)    # Ground Type: Ground Plane（y = 0 に地面を置く）
67
68    out = geo.createNode("null", "OUT")
69    out.setFirstInput(solver)
70    finish(geo, out)
71    return out
72
73
74def main() -> None:
75    out = build()
76    check_errors(out.parent())
77    # フレームを進めて、箱の重心の高さが変わることを確かめる。
78    for frame in (1, 24, 48):
79        hou.setFrame(frame)
80        heights = [point.position()[1] for point in out.geometry().points()]
81        print(f"フレーム {frame:2d}: ポイントの高さの平均 {sum(heights) / len(heights):.3f} m")
82    save_hip_from_argv()
83
84
85if __name__ == "__main__":
86    main()
